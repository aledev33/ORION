# requirements-rpi.txt

# ORION — dependencias para Raspberry Pi 4 (aarch64, Debian Trixie)
#
# Instalar con:
#   pip install -r requirements-rpi.txt --break-system-packages
#
# Paquetes del sistema necesarios (instalar primero con apt):
#   sudo apt install -y \
#       espeak-ng \
#       portaudio19-dev \
#       python3-pyaudio \
#       alsa-utils \
#       pulseaudio \
#       scrot \
#       libgirepository1.0-dev \
#       i2c-tools \
#       python3-smbus
#
# Para la pantalla OLED SSD1306 (I2C):
#   sudo apt install -y python3-pil python3-pil.imagetk
#   sudo raspi-config → Interface Options → I2C → Enable
#
